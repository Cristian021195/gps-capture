import { useEffect, useMemo, useRef, useState } from "react";
import { Accordion } from "../ui/Accordion"
import { Button, ListInput, List } from "konsta/react";
import { ClockIcon, ClockRemoveIcon, DestinoIcon, DiasIcon, HorarioFormMenuIcon, OrigenIcon, QRShareIcon, RutaIcon, DineroIcon } from '../svg/FormIcons';
import { InfoIcon } from "../svg/InfoIcon";
import { ArrowClose, ArrowOpen, ArrowSwitch } from "../svg/Arrows";
import { useIntl } from "react-intl";
import { DateTimeHorario } from "../../classes/DateTimeHorario";
import { useHorarioFilter } from "../../hooks/useHorarioFilter";
import { useConfig } from "../../store/config";
import { useNavigate, useParams } from "react-router-dom";
import { useDBDatosEmpresa, useDBDatosEmpresaUrl } from "../../hooks/useDBDatosEmpresa";
import { useDBHorariosEmpresa } from "../../hooks/useDBHorariosEmpresa";
import { Horario } from "../../classes/Horario";
import type { IHorarioResultado } from "../../interfaces/IHorario";
import { SegmentedChoiceButton } from "./SegmentedChoiceButton";
import { TablaHorario } from "../ui/TablaHorario";
import { useBottomModal } from "../../store/bottom_modal";
import { BottomModalHorarioQR } from "../floating/BottomModalHorarioQR";
import { usePopUp } from "../../store/popup";
import { PopupHorarioInfo } from "../floating/PopupHorarioInfo";
import { useNavbar } from "../../store/navbar";
import { SubtitlePill } from "../ui/NavbarSubtitle";
import { GLOBAL } from "../../constants/global";
import { useUpdateSearchParams } from "../../hooks/useUpdateSearchParams";
import { PopupHorarioDescripcion } from "../floating/PopupHorarioDescripcion";
import { BottomModalHorarioOpciones } from "../floating/BottomModalHorarioOpciones";
import { MatrizHelper } from '../../classes/MatrizHelper';

export const HorarioForm = () => {
    const { id } = useParams();
    const [opn, setOpn] = useState(true);
    const [isLoading] = useState(false);
    const formRef = useRef<HTMLFormElement>(null);
    //const searchRef = useRef<HTMLFormElement>(null);
    const {formatMessage:tr} = useIntl();//grid grid-cols-2 horarios-step-0
    const {schedule_filters, setScheduleAuto} = useConfig();
    const {setSubtitle, setRight} = useNavbar();
    const navigate = useNavigate();
    const updateParams = useUpdateSearchParams();
    const {filtros, handleChangeFilter, clearTime, switchCiudades, resetFilters} = useHorarioFilter(id);//id
    const {rutas} = useDBDatosEmpresa(id);
    const {horario_query_result} = useDBHorariosEmpresa(filtros.key);
    const {empresa_url_query_result} = useDBDatosEmpresaUrl(filtros.key);
    const [busquedaPrecio, setBusquedaPrecio] = useState(false);
    
    //const horario = horario_query_result ? new Horario(horario_query_result) : null;
    const horario = useMemo(() => {
        return horario_query_result ? new Horario(horario_query_result) : null;
    }, [horario_query_result]);
    const [ciudades, setCiudades] = useState<string[]>(horario?.ciudades || []);
    const [ciudades_destino, setCiudadesDestino] = useState<string[]>(horario ? [...horario.ciudades].reverse() : []);
    const [resultado, setResultado] = useState<IHorarioResultado>();
    const [selectedSegmentedPrecio, setSelectedSegmentedPrecio] = useState<number[]>([]);
    const {setBottomModal} = useBottomModal();
    const {setPopUp} = usePopUp();

    useEffect(() => {
        if (horario && !MatrizHelper.arraysIguales(ciudades, horario.ciudades)) {
            if(horario.horario.sentido === 1){
                // eslint-disable-next-line react-hooks/set-state-in-effect
                setCiudadesDestino([...horario.ciudades].reverse());
                setCiudades(horario.ciudades);
            }else{
                setCiudades([...horario.ciudades].reverse());
                setCiudadesDestino(horario.ciudades);
            }
            
            setRight(<button className="px-2.5 hithere-3" onClick={()=>{
                updateParams({ emergent: "popupbox"});
                setPopUp({
                    title:tr({id:'info.horario'}),
                    children: <PopupHorarioDescripcion description={horario_query_result?.descripcion}/>
                })
            }}>
                <InfoIcon size={24}/>
            </button>)
        }
    }, [horario]);

    useEffect(() => {        
        if (horario) {
            // eslint-disable-next-line react-hooks/set-state-in-effect
            setResultado(horario.getHorarioByStr(filtros.origen, filtros.destino,filtros.origen_str, filtros.destino_str, filtros.hora));
            //setResultado(horario.getHorario(filtros.origen, filtros.destino, filtros.hora)); // bk 2-5-26 porlas           
            setSubtitle(<SubtitlePill className="atention" text={tr({id:'vigencia'})+": "+horario.getVigencia()}></SubtitlePill>);
        }else{
            setResultado(undefined);
        }
    }, [filtros, horario]);

    useEffect(()=>{
        if(resultado?.precio !== 0){
            // eslint-disable-next-line react-hooks/set-state-in-effect
            setBusquedaPrecio(false);
        }
    },[resultado?.precio])

    const value = useMemo(() => {
        if (selectedSegmentedPrecio.length !== 2){
            return null
        }else{
            return horario?.cuerpo_precio[selectedSegmentedPrecio[0]][selectedSegmentedPrecio[1]];
        }        
    }, [selectedSegmentedPrecio, horario]);

    /* rollback: dias de semana
        <optgroup label={tr({id:'comunes'})}>
            <option value="7">7 {tr({id:'hab'})}</option>
            <option value="5">5 {tr({id:'sab'})}</option> 
            <option value="6">6 {tr({id:'dom'})}</option>
        </optgroup>
        <optgroup label={tr({id:'especiales'})}>
            <option value="0">0 {tr({id:'lun'})}</option>
            <option value="1">1 {tr({id:'mar'})}</option>
            <option value="2">2 {tr({id:'mie'})}</option>
            <option value="3">3 {tr({id:'jue'})}</option>
            <option value="4">4 {tr({id:'vie'})}</option>
            <option value="8">8 {tr({id:'fer'})}</option>
            <option value="9">9 {tr({id:'esp'})}</option>
        </optgroup>
    */
    
    return <>
        <Accordion opn={opn}>
            <form ref={formRef}>
                <List strongIos insetIos className='my-1'>
                    <div className="grid grid-cols-2 horarios-step-0">
                        <ListInput
                            title={tr({id:'dias'})}
                            type="select"
                            dropdown
                            colors={{bgMaterial:'k-panel-form'}}
                            value={filtros.dias}
                            placeholder={DateTimeHorario.getDiaActualTr(filtros.dias)}
                            disabled={isLoading}
                            name="dias"
                            onChange={handleChangeFilter}
                            media={<DiasIcon />}>
                                <option value="0"> {tr({id:'hab'})}</option>
                                <option value="1"> {tr({id:'sab'})}</option>
                                <option value="2"> {tr({id:'dom'})}</option>
                                <option value="3"> {tr({id:'fer'})}</option>
                        </ListInput>
                        <ListInput
                            title={tr({id:'company.schedule.route'})}
                            type="select"
                            dropdown
                            colors={{bgMaterial:'k-panel-form'}}
                            value={filtros.ruta}
                            name="ruta"
                            className='horarios-step-1'
                            disabled={isLoading}
                            placeholder="Ruta"
                            onChange={handleChangeFilter}
                            media={<RutaIcon />}>
                                <optgroup>
                                    <option value="">{tr({id:'select'})+" "+tr({id:'company.schedule.route'})}</option>
                                </optgroup>
                                <optgroup>                                    
                                    {
                                        rutas.map(ruta=>{
                                            return <option key={ruta} value={ruta}>{ruta}</option>
                                        })
                                    }
                                </optgroup>
                        </ListInput>
                    </div>
                    <div className="grid grid-cols-2">
                        <ListInput
                            title={tr({id:'company.schedule.origin'})}
                            type="select"
                            dropdown
                            colors={{bgMaterial:'k-panel-form'}}
                            value={filtros.origen}
                            placeholder="Todos"
                            disabled={isLoading}
                            name="origen"
                            onChange={handleChangeFilter}
                            media={<OrigenIcon />}>
                                <optgroup>
                                    <option value="1">{tr({id:'ns'})}</option>
                                    <option value="0">{tr({id:'sn'})}</option>
                                    <option value="1">{tr({id:'oe'})}</option>
                                    <option value="0">{tr({id:'eo'})}</option>
                                </optgroup>                                
                                <optgroup className='[&_option]:capitalize'>
                                    {
                                        schedule_filters && ciudades.map((ciudad, ci)=><option key={ciudad} value={ci+2}>
                                            {ciudad}
                                        </option>)
                                    }
                                </optgroup>
                        </ListInput>
                        {
                            schedule_filters ? 
                            <ListInput
                                title={tr({id:'company.schedule.destination'})}
                                type="select"
                                dropdown
                                colors={{bgMaterial:'k-panel-form'}}
                                value={filtros.destino}
                                placeholder="Destino"
                                name="destino"
                                disabled={isLoading}
                                onChange={handleChangeFilter}
                                media={<DestinoIcon />}>
                                    <optgroup>
                                        <option value="">{tr({id:'select'})+" "+tr({id:'company.schedule.destination'})}</option>
                                    </optgroup>
                                    <optgroup className='[&_option]:capitalize'>
                                        {
                                            ciudades_destino.map((ciudad, ci)=><option key={ciudad} value={ciudades_destino.length-(ci-1)}>
                                                {ciudad}
                                            </option>)
                                        }
                                    </optgroup>
                            </ListInput> : 
                            <div className="my-2 mx-3.5 px-1 text-xl flex justify-center items-center">
                                {
                                    resultado && resultado.precio !== 0 && !schedule_filters
                                    && <div className="flex gap-2 items-center text-3xl hithere-3">
                                        <DineroIcon width={24} height={24}/> {
                                            (resultado.precio! > 0 || value! > 0) ? resultado.precio || value : <small className="text-sm">{tr({id:'pending'})}</small>
                                        }
                                    </div>
                                }
                            </div>
                        }
                    </div>
                    {
                        schedule_filters 
                            ? <div className="grid grid-cols-2 horarios-step-1">
                                <ListInput
                                    title={tr({id:'company.schedule.time'})}
                                    disabled={isLoading}
                                    type="time"
                                    name="hora"
                                    colors={{bgMaterial:'k-panel-form'}}
                                    placeholder="Please choose..."
                                    clearButton={true}
                                    value={filtros.hora}
                                    onClear={()=>{
                                        clearTime();
                                        setScheduleAuto(false);
                                    }}
                                    media={<ClockIcon/>}
                                    onChange={handleChangeFilter}>
                                </ListInput>
                            <div className="my-2 mx-3.5 px-1 text-xl space-y-2 flex justify-center items-center">
                                {
                                    resultado && (resultado.precio !== 0 || value) && schedule_filters
                                    ? resultado.precio !== null && <div className="flex gap-2 items-center text-3xl">
                                        <DineroIcon key={resultado.precio} width={24} height={24} className="hithere"/>{
                                            (resultado.precio > 0 || value! > 0) ? resultado.precio || value : <small className="text-sm">{tr({id:'pending'})}</small>
                                        }
                                    </div>
                                    : (horario?.ciudades_precio?.length ?? 0) > 0 && <div>
                                        <div className="text-base">{tr({id:'select'})} ({selectedSegmentedPrecio.length}/2)</div>
                                        <Button type="button" small className="w-fit mx-auto k-btn-tonal" disabled={!resultado} onClick={()=>{setBusquedaPrecio(p=>!p)}}>{tr({id:'ver.precios'})}</Button>                                        
                                    </div>
                                }
                            </div>
                        </div>
                        : <></>
                    }
                    {
                        resultado && schedule_filters && busquedaPrecio &&
                        <SegmentedChoiceButton className="mx-4 mt-1.5 ios:py-3" ciudades={horario?.ciudades_precio} selected={selectedSegmentedPrecio} onChange={setSelectedSegmentedPrecio}/>
                    }
                </List>    
            </form>
        </Accordion>
        <div className="mx-4">

            <div className="flex gap-3">
                <Button title="Actualizar" className="k-btn-tonal" onClick={()=>{
                    updateParams({ emergent: "bottommodal" });
                    setBottomModal({title:tr({id:'opciones'}), children:<BottomModalHorarioOpciones url={empresa_url_query_result} empresa_id={horario_query_result?.empresa_id} resetFilters={resetFilters}/>})
                }} disabled={isLoading}>
                    <HorarioFormMenuIcon/> 
                </Button>
                <Button title='Compartir código QR' className="k-btn-tonal" disabled={isLoading} onClick={
                    ()=>{
                        updateParams({ emergent: "bottommodal" });
                        setBottomModal({title:tr({id:'scan.title'}), children:<BottomModalHorarioQR urls={GLOBAL.URL_EMPRESAS+"|"+empresa_url_query_result?.replace(GLOBAL.APP_SCRIPT_BASE_URL, "")} cb={()=>{}}/>})
                    }
                }>
                    <QRShareIcon/>
                </Button>
                {
                    schedule_filters ? 
                    <Button title='Cambiar origen y destino' className="k-btn-tonal" onClick={()=>{
                        switchCiudades()
                    }}>
                        <ArrowSwitch/>
                    </Button> : 
                    <Button title='Filtrado tiempo actual' className="k-btn-tonal" disabled={isLoading} onClick={()=>{
                    }}>
                        {
                            schedule_filters ? 
                            <ClockIcon/> : 
                            <ClockRemoveIcon/>
                        }
                    </Button>
                }
                <Button title='Info' className="k-btn-tonal" onClick={()=>{
                    updateParams({ emergent: "popupbox"});
                    setPopUp({
                        title:tr({id:'info'}),
                        children: <PopupHorarioInfo cb={()=>{
                            updateParams({ emergent: null }, { replace: true });
                            navigate('/info');
                        }}/>
                    })
                }}>
                    <InfoIcon/>
                </Button>
                <Button title='Colapsar' className="k-btn-tonal" disabled={isLoading} onClick={()=>{setOpn(!opn)}}>{opn ? <ArrowClose/> : <ArrowOpen/>} </Button>
            </div>
            <div className="mt-3">
                <TablaHorario collapsed={opn} resultado_horario={resultado} empresa={horario?.getEmpresa()} cabecera={ciudades} precio={resultado?.precio} precios={horario?.getPreciosBody()}/>
            </div>
        </div>
    </>
}
